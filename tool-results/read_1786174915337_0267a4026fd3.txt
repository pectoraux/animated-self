     3→/**
     4→ * Phase 3 — Async Render panel.
     5→ *
     6→ * The high-fidelity / offline path. The user picks a character (same shared
     7→ * store as the Studio picker), supplies an audio or video driver, picks a
     8→ * quality tier, and submits a render job to POST /api/render. The panel polls
     9→ * GET /api/render/{job_id} every 500ms and shows status / progress / download.
    10→ *
    11→ * Consent gate (same as live): a bound custom character (consented=true,
    12→ * source=generated/uploaded) requires a fresh consent_token from liveness.
    13→ * The panel surfaces a "Re-run liveness to render" CTA that opens the shared
    14→ * LivenessDialog; on success the captured token is sent with the render
    15→ * request. Stock characters need no token.
    16→ *
    17→ * Demo mode (engine not connected — the sandbox state): the panel shares
    18→ * `engineConnected` from the Zustand store (set by the Studio + Create panel
    19→ * probes). When false, the Render button still works but simulates the entire
    20→ * job lifecycle (queued → running → rendering → done) over ~3s with a
    21→ * "Simulated" badge, and the download button is disabled with an honest note.
    22→ */

    70→// ---------------------------------------------------------------------------
    71→// Types
    72→// ---------------------------------------------------------------------------
    73→
    74→type DriverKind = "audio" | "video";
    75→type DriverMode = "upload" | "url";
    76→type Quality = "draft" | "high";
    77→
    78→interface ActiveJob {
    79→  jobId: string;
    80→  status: RenderJobStatus;
    81→  progress: number; // 0..1
    82→  error: string | null;
    83→  startedAt: number;
    84→  /** Wall-clock ms from job start to terminal state; null while in flight. */
    85→  elapsedMs: number | null;
    86→  /** True when running the demo-mode simulation (engine not connected). */
    87→  demo: boolean;
    88→}
    89→
    90→// ---------------------------------------------------------------------------
    91→// Component
    92→// ---------------------------------------------------------------------------
    93→
    94→export function AsyncRender() {
    95→  const characters = useCharacterStore((s) => s.characters);
    96→  const selectedId = useCharacterStore((s) => s.selectedId);
    97→  const setSelected = useCharacterStore((s) => s.setSelected);
    98→  const engineConnected = useCharacterStore((s) => s.engineConnected);
    99→
   100→  const [driverKind, setDriverKind] = React.useState<DriverKind>("audio");
   101→  const [driverMode, setDriverMode] = React.useState<DriverMode>("url");
   102→  const [driverFile, setDriverFile] = React.useState<File | null>(null);
   103→  const [driverUrl, setDriverUrl] = React.useState("");
   104→  const [quality, setQuality] = React.useState<Quality>("draft");
   105→
   106→  /** Fresh consent_token from liveness; required to render a bound char. */
   107→  const [consentToken, setConsentToken] = React.useState<string | null>(null);
   108→  const [activeLiveness, setActiveLiveness] = React.useState(false);
   109→
   110→  const [job, setJob] = React.useState<ActiveJob | null>(null);
   111→  const [busy, setBusy] = React.useState(false);
   112→  const [error, setError] = React.useState<string | null>(null);
   113→
   114→  const pollRef = React.useRef<ReturnType<typeof setInterval> | null>(null);
   115→  const demoTimerRef = React.useRef<ReturnType<typeof setInterval> | null>(
   116→    null,
   117→  );
   118→
   119→  // Derive the effective driver_url from the active sub-mode.
   120→  const driverUrlValue =
   121→    driverMode === "upload"
   122→      ? driverFile
   123→        ? `upload://${driverFile.name}`
   124→        : ""
   125→      : driverUrl.trim();
   126→
   127→  const selected =
   128→    characters.find((c) => c.id === selectedId) ?? characters[0]!;
   129→
   130→  const isLocked = selected.source !== "stock" && !selected.consented;
   131→  const isBoundCustom =
   132→    selected.source !== "stock" && selected.consented;
   133→  const needsLiveness = isBoundCustom && !consentToken;
   134→
   135→  const jobInFlight =
   136→    job !== null &&
   137→    (job.status === "queued" ||
   138→      job.status === "running" ||
   139→      job.status === "rendering");
   140→
   141→  const canRender =
   142→    !busy &&
   143→    !jobInFlight &&
   144→    driverUrlValue.length > 0 &&
   145→    !isLocked &&
   146→    !needsLiveness;
   147→
   148→  // Clear the captured token + error when the user picks a different
   149→  // character — the token is face-specific, not char-specific, but clearing
   150→  // keeps the mental model simple ("re-run liveness per render session").
   151→  React.useEffect(() => {
   152→    setConsentToken(null);
   153→    setError(null);
   154→  }, [selectedId]);
   155→
   156→  // --- Polling for live jobs ---------------------------------------------
   157→  React.useEffect(() => {
   158→    if (!job || job.demo) return;
   159→    if (job.status === "done" || job.status === "failed") return;
   160→
   161→    const id = job.jobId;
   162→    pollRef.current = setInterval(async () => {
   163→      try {
   164→        const res = await fetch(renderJobUrl(id));
   165→        if (!res.ok) {
   166→          throw new Error(`poll returned ${res.status}`);
   167→        }
   168→        const j = (await res.json()) as RenderJob;
   169→        setJob((prev) => {
   170→          if (!prev || prev.jobId !== id) return prev;
   171→          const terminal =
   172→            j.status === "done" || j.status === "failed";
   173→          return {
   174→            ...prev,
   175→            status: j.status,
   176→            progress: j.progress,
   177→            error: j.error ?? null,
   178→            elapsedMs: terminal ? Date.now() - prev.startedAt : prev.elapsedMs,
   179→          };
   180→        });
   181→      } catch {
   182→        // Transient network hiccup — keep the last known state. The next
   183→        // tick will retry.
   184→      }
   185→    }, 500);
   186→
   187→    return () => {
   188→      if (pollRef.current) {
   189→        clearInterval(pollRef.current);
   190→        pollRef.current = null;
   191→      }
   192→    };
   193→  }, [job?.jobId, job?.status, job?.demo]);
   194→
   195→  // --- Cleanup on unmount -------------------------------------------------
   196→  React.useEffect(
   197→    () => () => {
   198→      if (pollRef.current) clearInterval(pollRef.current);
   199→      if (demoTimerRef.current) clearInterval(demoTimerRef.current);
   200→    },
   201→    [],
   202→  );
   203→
   204→  // --- Actions ------------------------------------------------------------
   205→
   206→  const onRender = async () => {
   207→    if (!canRender) return;
   208→    setError(null);
   209→    setBusy(true);
   210→    setJob(null);
   211→
   212→    // Clear any prior demo timer before starting fresh.
   213→    if (demoTimerRef.current) {
   214→      clearInterval(demoTimerRef.current);
   215→      demoTimerRef.current = null;
   216→    }
   217→
   218→    const body: RenderRequest = {
   219→      character_id: selected.id,
   220→      driver: driverKind,
   221→      driver_url: driverUrlValue,
   222→      quality,
   223→      consent_token: isBoundCustom ? consentToken : null,
   224→    };
   225→
   226→    if (!engineConnected) {
   227→      // --- Demo simulation ---
   228→      const demoJobId = `demo-${Math.random().toString(36).slice(2, 8)}`;
   229→      const startedAt = Date.now();
   230→      setJob({
   231→        jobId: demoJobId,
   232→        status: "queued",
   233→        progress: 0,
   234→        error: null,
   235→        startedAt,
   236→        elapsedMs: null,
   237→        demo: true,
   238→      });
   239→      setBusy(false);
   240→
   241→      // Walk the lifecycle: queued -> running -> rendering -> done over ~3s.
   242→      setTimeout(() => {
   243→        setJob((prev) =>
   244→          prev && prev.demo
   245→            ? { ...prev, status: "running", progress: 0.05 }
   246→            : prev,
   247→        );
   248→      }, 350);
   249→      setTimeout(() => {
   250→        setJob((prev) =>
   251→          prev && prev.demo
   252→            ? { ...prev, status: "rendering", progress: 0.2 }
   253→            : prev,
   254→        );
   255→      }, 750);
   256→
   257→      demoTimerRef.current = setInterval(() => {
   258→        setJob((prev) => {
   259→          if (!prev || !prev.demo) return prev;
   260→          const next = Math.min(
   261→            1,
   262→            prev.progress + 0.05 + Math.random() * 0.04,
   263→          );
   264→          if (next >= 1) {
   265→            if (demoTimerRef.current) {
   266→              clearInterval(demoTimerRef.current);
   267→              demoTimerRef.current = null;
   268→            }
   269→            return {
   270→              ...prev,
   271→              progress: 1,
   272→              status: "done",
   273→              elapsedMs: Date.now() - prev.startedAt,
   274→            };
   275→          }
   276→          return { ...prev, progress: next };
   277→        });
   278→      }, 160);
   279→      return;
   280→    }
   281→
   282→    // --- Live render ---
   283→    try {
   284→      const res = await fetch(engineUrl("/api/render"), {
   285→        method: "POST",
   286→        headers: { "Content-Type": "application/json" },
   287→        body: JSON.stringify(body),
   288→      });
   289→      if (!res.ok) {
   290→        const txt = await res.text().catch(() => "");
   291→        throw new Error(
   292→          `Engine returned ${res.status}${txt ? `: ${txt.slice(0, 200)}` : ""}`,
   293→        );
   294→      }
   295→      const j = (await res.json()) as RenderJob;
   296→      setJob({
   297→        jobId: j.job_id,
   298→        status: j.status,
   299→        progress: j.progress,
   300→        error: j.error ?? null,
   301→        startedAt: Date.now(),
   302→        elapsedMs: null,
   303→        demo: false,
   304→      });
   305→    } catch (e) {
   306→      setError(e instanceof Error ? e.message : String(e));
   307→    } finally {
   308→      setBusy(false);
   309→    }
   310→  };
   311→
   312→  const onLivenessBound = (_id: string, token: string) => {
   313→    setConsentToken(token);
   314→    setActiveLiveness(false);
   315→  };
   316→
   317→  const onResetJob = () => {
   318→    if (pollRef.current) {
   319→      clearInterval(pollRef.current);
   320→      pollRef.current = null;
   321→    }
   322→    if (demoTimerRef.current) {
   323→      clearInterval(demoTimerRef.current);
   324→      demoTimerRef.current = null;
   325→    }
   326→    setJob(null);
   327→    // Require fresh liveness for the next render (tokens expire).
   328→    setConsentToken(null);
   329→  };
   330→
   331→  // --- Render -------------------------------------------------------------
   332→  return (
   333→    <Card className="overflow-hidden border-neutral-800 bg-neutral-900/60">
   334→      <CardHeader className="border-b border-neutral-800">
   335→        <div className="flex flex-wrap items-center justify-between gap-3">
   336→          <div>
   337→            <CardTitle className="flex items-center gap-2 text-neutral-50">
   338→              <Film className="size-5 text-rose-300" />
   339→              Async render — quality mode
   340→            </CardTitle>
   341→            <CardDescription className="text-neutral-400">
   342→              Phase 3 — offline diffusion-grade renders. Submit a job, watch
   343→              the queue, download the MP4 when it&rsquo;s done.
   344→            </CardDescription>
   345→          </div>
   346→          <EngineBadge connected={engineConnected} />
   347→        </div>
   348→      </CardHeader>
   349→
   350→      <CardContent className="grid gap-6 p-6 lg:grid-cols-[1fr_minmax(0,360px)]">
   351→        {/* LEFT — controls */}
   352→        <div className="space-y-6">
   353→          {/* Character picker */}
   354→          <div>
   355→            <div className="mb-2 flex items-center justify-between">
   356→              <h3 className="text-xs font-semibold uppercase tracking-wider text-neutral-400">
   357→                Character
   358→              </h3>
   359→              <span className="font-mono text-[11px] text-neutral-500">
   360→                {characters.length} available
   361→                {characters.some((c) => !c.consented && c.source !== "stock")
   362→                  ? " · some locked"
   363→                  : ""}
   364→              </span>
   365→            </div>
   366→            <div className="grid grid-cols-3 gap-2">
   367→              {characters.map((c) => {
   368→                const active = c.id === selected.id;
   369→                const locked =
   370→                  !c.consented && c.source !== "stock";
   371→                return (
   372→                  <CharacterPickerButton
   373→                    key={c.id}
   374→                    character={c}
   375→                    active={active}
   376→                    locked={locked}
   377→                    onPick={() => {
   378→                      if (!locked) setSelected(c.id);
   379→                    }}
   380→                  />
   381→                );
   382→              })}
   383→            </div>
   384→            {characters.some((c) => !c.consented && c.source !== "stock") ? (
   385→              <p className="mt-2 text-[11px] text-amber-200/80">
   386→                Locked characters need a liveness check first — use the Create
   387→                Character panel above to unlock them.
   388→              </p>
   389→            ) : null}
   390→          </div>
   391→
   392→          {/* Driver input */}
   393→          <div>
   394→            <h3 className="mb-2 text-xs font-semibold uppercase tracking-wider text-neutral-400">
   395→              Driver
   396→            </h3>
   397→            <Tabs
   398→              value={driverKind}
   399→              onValueChange={(v) => setDriverKind(v as DriverKind)}
   400→              className="gap-3"
   401→            >
   402→              <TabsList className="bg-neutral-950/60">
   403→                <TabsTrigger
   404→                  value="audio"
   405→                  className="data-[state=active]:bg-rose-500/15 data-[state=active]:text-rose-200"
   406→                >
   407→                  <AudioLines className="size-3.5" />
   408→                  Audio
   409→                </TabsTrigger>
   410→                <TabsTrigger
   411→                  value="video"
   412→                  className="data-[state=active]:bg-rose-500/15 data-[state=active]:text-rose-200"
   413→                >
   414→                  <Video className="size-3.5" />
   415→                  Video
   416→                </TabsTrigger>
   417→              </TabsList>
   418→
   419→              <TabsContent value="audio" className="space-y-3">
   420→                <DriverInputRow
   421→                  kind="audio"
   422→                  mode={driverMode}
   423→                  onModeChange={setDriverMode}
   424→                  file={driverFile}
   425→                  onFile={setDriverFile}
   426→                  url={driverUrl}
   427→                  onUrl={setDriverUrl}
   428→                />
   429→                <p className="text-[11px] leading-relaxed text-neutral-500">
   430→                  Audio-driven lip-sync. The engine runs whatever renderer you
   431→                  configure via <code className="text-neutral-400">DIFFUSION_RENDER_CMD</code>
   432→                  (an AniPortrait-style setup, or anything honoring the same
   433→                  reference-image + audio-in, MP4-out contract). No model is
   434→                  bundled — see <code className="text-neutral-400">docs/reality-check.md</code>.
   435→                </p>
   436→              </TabsContent>
   437→
   438→              <TabsContent value="video" className="space-y-3">
   439→                <div className="rounded-lg border border-neutral-800 bg-neutral-950/40 p-4 text-center">
   440→                  <p className="text-sm text-neutral-400">
   441→                    Video re-drive is not implemented yet.
   442→                  </p>
   443→                  <p className="mt-1 text-[11px] text-neutral-500">
   444→                    The engine currently accepts <code className="text-neutral-400">driver="audio"</code> only.
   445→                    Video re-drive (transferring pose + expression from a recorded
   446→                    video) is scoped for a later phase.
   447→                  </p>
   448→                </div>
   449→              </TabsContent>
   450→            </Tabs>
   451→          </div>
   452→
   453→          {/* Quality selector */}
   454→          <div>
   455→            <h3 className="mb-2 text-xs font-semibold uppercase tracking-wider text-neutral-400">
   456→              Quality
   457→            </h3>
   458→            <div className="inline-flex rounded-lg border border-neutral-800 bg-neutral-950/60 p-1">
   459→              <QualityToggle
   460→                active={quality === "draft"}
   461→                onClick={() => setQuality("draft")}
   462→                label="Draft"
   463→                hint="fast"
   464→              />
   465→              <QualityToggle
   466→                active={quality === "high"}
   467→                onClick={() => setQuality("high")}
   468→                label="High"
   469→                hint="diffusion"
   470→              />
   471→            </div>
   472→            <p className="mt-2 text-xs text-neutral-500">
   473→              The quality field is reserved for when multiple renderer backends
   474→              exist. Currently all renders go through the single{" "}
   475→              <code className="text-neutral-400">DIFFUSION_RENDER_CMD</code>{" "}
   476→              command regardless of this setting.
   477→            </p>
   478→          </div>
   479→
   480→          {/* Action area */}
   481→          <div className="space-y-3">
   482→            {isLocked ? (
   483→              <Alert className="border-amber-500/40 bg-amber-500/10 text-amber-200">
   484→                <Lock className="size-4" />
   485→                <AlertTitle>Character is locked</AlertTitle>
   486→                <AlertDescription className="text-amber-100/80">
   487→                  Complete liveness for this character in the Create Character
   488→                  panel above before you can render it.
   489→                </AlertDescription>
   490→              </Alert>
   491→            ) : needsLiveness ? (
   492→              <div className="space-y-3">
   493→                <Alert className="border-rose-500/30 bg-rose-500/[0.06] text-rose-100">
   494→                  <ScanFace className="size-4 text-rose-300" />
   495→                  <AlertTitle className="text-rose-200">
   496→                    This character is bound to your face
   497→                  </AlertTitle>
   498→                  <AlertDescription className="text-neutral-300">
   499→                    Async renders for a bound character require a fresh
   500→                    consent_token. Liveness tokens expire, so re-run the check
   501→                    to mint a new one for this render.
   502→                  </AlertDescription>
   503→                </Alert>
   504→                <Button
   505→                  type="button"
   506→                  size="lg"
   507→                  onClick={() => setActiveLiveness(true)}
   508→                  className="w-full bg-rose-500 text-white hover:bg-rose-400"
   509→                >
   510→                  <ScanFace className="size-4" />
   511→                  Re-run liveness to render
   512→                </Button>
   513→                {consentToken ? (
   514→                  <p className="text-[11px] text-emerald-300">
   515→                    Token captured — you can now render.
   516→                  </p>
   517→                ) : null}
   518→              </div>
   519→            ) : consentToken ? (
   520→              <div className="space-y-2">
   521→                <Button
   522→                  type="button"
   523→                  size="lg"
   524→                  onClick={onRender}
   525→                  disabled={!canRender}
   526→                  className="w-full bg-rose-500 text-white hover:bg-rose-400 disabled:cursor-not-allowed disabled:opacity-50"
   527→                >
   528→                  {busy ? (
   529→                    <Loader2 className="size-4 animate-spin" />
   530→                  ) : (
   531→                    <Film className="size-4" />
   532→                  )}
   533→                  {busy ? "Submitting…" : "Render with captured token"}
   534→                </Button>
   535→                <p className="text-[11px] text-emerald-300">
   536→                  Liveness verified — consent_token ready.
   537→                </p>
   538→              </div>
   539→            ) : (
   540→              <Button
   541→                type="button"
   542→                size="lg"
   543→                onClick={onRender}
   544→                disabled={!canRender}
   545→                className="w-full bg-rose-500 text-white hover:bg-rose-400 disabled:cursor-not-allowed disabled:opacity-50"
   546→              >
   547→                {busy ? (
   548→                  <Loader2 className="size-4 animate-spin" />
   549→                ) : (
   550→                  <Film className="size-4" />
   551→                )}
   552→                {busy ? "Submitting…" : "Render"}
   553→              </Button>
   554→            )}
   555→
   556→            {!engineConnected ? (
   557→              <p className="text-[11px] text-amber-200/80">
   558→                Demo mode — clicking Render simulates a job (queued → running →
   559→                rendering → done over ~3s). No real file is produced.
   560→              </p>
   561→            ) : null}
   562→          </div>
   563→
   564→          {/* Error */}
   565→          {error ? (
   566→            <Alert className="border-amber-500/40 bg-amber-500/10 text-amber-200">
   567→              <CircleAlert className="size-4" />
   568→              <AlertTitle>Render failed</AlertTitle>
   569→              <AlertDescription className="text-amber-100/80">
   570→                {error}
   571→              </AlertDescription>
   572→            </Alert>
   573→          ) : null}
   574→        </div>
   575→
   576→        {/* RIGHT — job progress / download */}
   577→        <div className="space-y-4">
   578→          <JobPanel
   579→            job={job}
   580→            onReset={onResetJob}
   581→            engineConnected={engineConnected}
   582→          />
   583→        </div>
   584→      </CardContent>
   585→
   586→      <LivenessDialog
   587→        characterId={activeLiveness ? selected.id : null}
   588→        isDemo={!engineConnected}
   589→        onClose={() => setActiveLiveness(false)}
   590→        onBound={onLivenessBound}
   591→      />
   592→    </Card>
   593→  );
   594→}

   599→
   600→function EngineBadge({ connected }: { connected: boolean }) {
   601→  if (connected) {
   602→    return (
   603→      <Badge
   604→        variant="outline"
   605→        className="gap-1.5 border-emerald-500/30 bg-emerald-500/10 text-emerald-300"
   606→      >
   607→        <CheckCircle2 className="size-3.5" />
   608→        Engine connected · live render queue
   609→      </Badge>
   610→    );

  1153→function formatBytes(bytes: number): string {
  1154→  if (bytes < 1024) return `${bytes} B`;
  1155→  if (bytes < 1024 * 1024) return `${(bytes / 1024).toFixed(1)} KB`;
  1156→  return `${(bytes / 1024 / 1024).toFixed(1)} MB`;
  1157→}
  1158→
  1159→function formatMs(ms: number): string {
  1160→  if (ms < 1000) return `${Math.round(ms)} ms`;
  1161→  const s = ms / 1000;
  1162→  if (s < 60) return `${s.toFixed(1)} s`;
  1163→  const m = Math.floor(s / 60);
  1164→  const rem = Math.round(s % 60);
  1165→  return `${m}m ${rem}s`;
  1166→}

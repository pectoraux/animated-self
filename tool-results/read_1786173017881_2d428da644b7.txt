   236→function useEngineProbe(): EngineProbe {
   237→  const [state, setState] = React.useState<EngineProbe>({ status: "probing" });
   238→
   239→  React.useEffect(() => {
   240→    let cancelled = false;
   241→    const ctrl = new AbortController();
   242→    const timer = setTimeout(() => ctrl.abort(), HEALTH_TIMEOUT_MS);
   243→
   244→    fetch(`/api/health?XTransformPort=${ENGINE_PORT}`, {
   245→      signal: ctrl.signal,
   246→    })
   247→      .then((r) =>
   248→        r.ok ? r.json() : Promise.reject(new Error(String(r.status))),
   249→      )
   250→      .then(() => {
   251→        if (!cancelled) setState({ status: "live" });
   252→      })
   253→      .catch(() => {
   254→        if (!cancelled) setState({ status: "demo" });
   255→      })
   256→      .finally(() => clearTimeout(timer));
   257→
   258→    return () => {
   259→      cancelled = true;
   260→      ctrl.abort();
   261→    };
   262→  }, []);
   263→
   264→  return state;
   265→}

   305→function useProviders(live: boolean) {
   306→  const [providers, setProviders] = React.useState<GenProviderInfo[]>([
   307→    { id: "demo", byok: false, requires_key: false, label: "Demo (no key)" },
   308→  ]);
   309→
   310→  React.useEffect(() => {
   311→    if (!live) {
   312→      setProviders([
   313→        {
   314→          id: "demo",
   315→          byok: false,
   316→          requires_key: false,
   317→          label: "Demo (no key) — simulated",
   318→        },
   319→        {
   320→          id: "openai",
   321→          byok: true,
   322→          requires_key: true,
   323→          label: "OpenAI DALL·E 3 (BYOK)",
   324→        },
   325→      ]);
   326→      return;
   327→    }
   328→    let cancelled = false;
   329→    fetch(engineUrl("/api/characters/providers"))
   330→      .then((r) => (r.ok ? r.json() : Promise.reject(new Error(String(r.status)))))
   331→      .then((list: GenProviderInfo[]) => {
   332→        if (cancelled || !Array.isArray(list) || list.length === 0) return;
   333→        setProviders(list);
   334→      })
   335→      .catch(() => {
   336→        if (cancelled) return;
   337→        // Fall back to the static list if the engine hiccups.
   338→        setProviders([
   339→          {
   340→            id: "demo",
   341→            byok: false,
   342→            requires_key: false,
   343→            label: "Demo (no key)",
   344→          },
   345→        ]);
   346→      });
   347→    return () => {
   348→      cancelled = true;
   349→    };
   350→  }, [live]);
   351→
   352→  return { providers };
   353→}

   435→function TextTab({ isDemo, providers, onCreated }: TabProps) {
   436→  const [name, setName] = React.useState("");
   437→  const [prompt, setPrompt] = React.useState(
   438→    "Anime girl with rose-pink hair, soft pastel palette, head-and-shoulders portrait, neutral expression, plain background",
   439→  );
   440→  const [providerId, setProviderId] = React.useState(
   441→    providers[0]?.id ?? "demo",
   442→  );
   443→  const [apiKey, setApiKey] = React.useState("");
   444→  const [busy, setBusy] = React.useState(false);
   445→  const [error, setError] = React.useState<string | null>(null);
   446→  const addCreated = useCharacterStore((s) => s.addCreated);
   447→
   448→  const provider = providers.find((p) => p.id === providerId);
   449→  const needsKey = provider?.requires_key === true;
   450→
   451→  React.useEffect(() => {
   452→    // If the provider list loads after first render, ensure selectedId exists.
   453→    if (!providers.some((p) => p.id === providerId)) {
   454→      setProviderId(providers[0]?.id ?? "demo");
   455→    }
   456→  }, [providers, providerId]);
   457→
   458→  const canSubmit =
   459→    !busy && name.trim().length > 0 && prompt.trim().length > 0 && (!needsKey || apiKey.trim().length > 0);
   460→
   461→  const onSubmit = async () => {
   462→    setError(null);
   463→    setBusy(true);
   464→    try {
   465→      if (!isDemo) {
   466→        const body: GenerateCharacterRequest = {
   467→          prompt: prompt.trim(),
   468→          name: name.trim(),
   469→          provider: providerId,
   470→          api_key: needsKey ? apiKey.trim() : null,
   471→        };
   472→        const ctrl = new AbortController();
   473→        const timer = setTimeout(() => ctrl.abort(), GEN_TIMEOUT_MS);
   474→        try {
   475→          const res = await fetch(engineUrl("/api/characters/generate"), {
   476→            method: "POST",
   477→            headers: { "Content-Type": "application/json" },
   478→            body: JSON.stringify(body),
   479→            signal: ctrl.signal,
   480→          });
   481→          if (!res.ok) {
   482→            const txt = await res.text().catch(() => "");
   483→            throw new Error(
   484→              `Engine returned ${res.status}${txt ? `: ${txt.slice(0, 200)}` : ""}`,
   485→            );
   486→          }
   487→          // Engine returns the new Character — but we don't strictly need its
   488→          // response: the next /api/characters poll would surface it. To keep
   489→          // the UX immediate we use addCreated with the local name + a tag.
   490→          const id = addCreated({
   491→            name: name.trim(),
   492→            source: "generated",
   493→            simulated: false,
   494→            tags: ["from-prompt"],
   495→          });
   496→          onCreated(id);
   497→        } finally {
   498→          clearTimeout(timer);
   499→        }
   500→      } else {
   501→        // Demo path: simulate ~2s of generation.
   502→        await sleep(2000);
   503→        const id = addCreated({
   504→          name: name.trim(),
   505→          source: "generated",
   506→          simulated: true,
   507→          tags: ["from-prompt"],
   508→        });
   509→        onCreated(id);
   510→      }
   511→    } catch (e) {
   512→      setError(e instanceof Error ? e.message : String(e));
   513→    } finally {
   514→      setBusy(false);
   515→    }
   516→  };
   517→
   518→  return (
   519→    <div className="grid gap-4">
   520→      <div className="grid gap-2">
   521→        <Label htmlFor="text-name" className="text-neutral-300">
   522→          Character name
   523→        </Label>
   524→        <Input
   525→          id="text-name"
   526→          placeholder="e.g. Hana"
   527→          value={name}
   528→          onChange={(e) => setName(e.target.value)}
   529→          className="bg-neutral-950/40"
   530→        />
   531→      </div>
   532→
   533→      <div className="grid gap-2">
   534→        <Label htmlFor="text-prompt" className="text-neutral-300">
   535→          Prompt
   536→        </Label>
   537→        <Textarea
   538→          id="text-prompt"
   539→          rows={4}
   540→          value={prompt}
   541→          onChange={(e) => setPrompt(e.target.value)}
   542→          className="bg-neutral-950/40 font-mono text-sm"
   543→          placeholder="Describe the character: hair, palette, expression, framing…"
   544→        />
   545→        <p className="text-[11px] text-neutral-500">
   546→          The engine forwards this verbatim to the provider&rsquo;s image model.
   547→        </p>
   548→      </div>
   549→
   550→      <div className="grid gap-4 sm:grid-cols-2">
   551→        <div className="grid gap-2">
   552→          <Label className="text-neutral-300">Provider</Label>
   553→          <ProviderSelect
   554→            value={providerId}
   555→            onValueChange={setProviderId}
   556→            providers={providers}
   557→          />
   558→        </div>
   559→        {needsKey ? (
   560→          <ApiKeyField
   561→            value={apiKey}
   562→            onChange={setApiKey}
   563→            providerLabel={provider?.label ?? "Provider"}
   564→          />
   565→        ) : null}
   566→      </div>
   567→
   568→      <SubmitRow
   569→        busy={busy}
   570→        disabled={!canSubmit}
   571→        onSubmit={onSubmit}
   572→        isDemo={isDemo}
   573→        actionLabel="Generate character"
   574→        icon={<Sparkles className="size-4" />}
   575→      />
   576→
   577→      {error ? (
   578→        <Alert className="border-amber-500/40 bg-amber-500/10 text-amber-200">
   579→          <CircleAlert className="size-4" />
   580→          <AlertTitle>Generation failed</AlertTitle>
   581→          <AlertDescription className="text-amber-100/80">
   582→            {error}
   583→          </AlertDescription>
   584→        </Alert>
   585→      ) : null}
   586→    </div>
   587→  );
   588→}

   594→function SelfieTab({ isDemo, providers, onCreated }: TabProps) {
   595→  const [name, setName] = React.useState("");
   596→  const [file, setFile] = React.useState<File | null>(null);
   597→  const [previewUrl, setPreviewUrl] = React.useState<string | null>(null);
   598→  const [providerId, setProviderId] = React.useState(
   599→    providers[0]?.id ?? "demo",
   600→  );
   601→  const [apiKey, setApiKey] = React.useState("");
   602→  const [busy, setBusy] = React.useState(false);
   603→  const [error, setError] = React.useState<string | null>(null);
   604→  const addCreated = useCharacterStore((s) => s.addCreated);
   605→
   606→  const provider = providers.find((p) => p.id === providerId);
   607→  const needsKey = provider?.requires_key === true;
   608→
   609→  React.useEffect(() => {
   610→    if (!providers.some((p) => p.id === providerId)) {
   611→      setProviderId(providers[0]?.id ?? "demo");
   612→    }
   613→  }, [providers, providerId]);
   614→
   615→  React.useEffect(() => {
   616→    if (!file) {
   617→      setPreviewUrl(null);
   618→      return;
   619→    }
   620→    const url = URL.createObjectURL(file);
   621→    setPreviewUrl(url);
   622→    return () => URL.revokeObjectURL(url);
   623→  }, [file]);
   624→
   625→  const canSubmit =
   626→    !busy &&
   627→    name.trim().length > 0 &&
   628→    file !== null &&
   629→    (!needsKey || apiKey.trim().length > 0);
   630→
   631→  const onSubmit = async () => {
   632→    if (!file) return;
   633→    setError(null);
   634→    setBusy(true);

  1363→async function requestChallenge(
  1364→  characterId: string,
  1365→  isDemo: boolean,
  1366→  setPhase: (p: LivenessPhase) => void,
  1367→) {
  1368→  setPhase({ kind: "requesting" });
  1369→  if (isDemo) {
  1370→    await sleep(700);
  1371→    const challenge: LivenessChallenge = {
  1372→      challenge_id: `demo-challenge-${Math.random().toString(36).slice(2, 8)}`,
  1373→      steps: DEMO_STEPS,
  1374→      issued_at: Date.now(),
  1375→    };
  1376→    setPhase({
  1377→      kind: "challenge",
  1378→      challenge,
  1379→      cameraStarted: false,
  1380→      verifying: false,
  1381→      progress: 0,
  1382→    });
  1383→    return;
  1384→  }
  1385→  try {
  1386→    const res = await fetch(engineUrl("/api/consent/liveness/request"), {
  1387→      method: "POST",
  1388→      headers: { "Content-Type": "application/json" },
  1389→      body: JSON.stringify({ character_id: characterId }),
  1390→    });
  1391→    if (!res.ok) {

  1395→    setPhase({
  1396→      kind: "challenge",

  1398→      cameraStarted: false,
  1399→      verifying: false,
  1400→      progress: 0,

  1402→  } catch (e) {

  1407→  }

  1410→async function verifyChallenge(
  1411→  characterId: string,
  1412→  challenge: LivenessChallenge,
  1413→  isDemo: boolean,
  1414→): Promise<LivenessResult> {
  1415→  // Match the engine's challenge steps back at it — in a real app this would
  1416→  // be the steps the client's MediaPipe actually detected.
  1417→  const detected = challenge.steps;
  1418→  const evidence = {
  1419→    landmark_count: 468,
  1420→    face_hash: `demo-${characterId}-${Date.now()}`,
  1421→  };
  1422→  if (isDemo) {
  1423→    await sleep(1500);
  1424→    return {
  1425→      challenge_id: challenge.challenge_id,
  1426→      passed: true,
  1427→      consent_token: `demo-token.${Math.random().toString(36).slice(2)}`,
  1428→    };
  1429→  }
  1430→  const res = await fetch(engineUrl("/api/consent/liveness/verify"), {
  1431→    method: "POST",
  1432→    headers: { "Content-Type": "application/json" },
  1433→    body: JSON.stringify({
  1434→      challenge_id: challenge.challenge_id,
  1435→      detected_steps: detected,
  1436→      landmark_evidence: evidence,
  1437→    }),
  1438→  });
  1439→  if (!res.ok) {
  1440→    throw new Error(`Verify returned ${res.status}`);
  1441→  }
  1442→  return (await res.json()) as LivenessResult;

  1510→function FileDrop({

  1613→function fileToDataUrl(file: File): Promise<string> {
  1614→  return new Promise((resolve, reject) => {
  1615→    const reader = new FileReader();
  1616→    reader.onload = () => resolve(reader.result as string);
  1617→    reader.onerror = () => reject(reader.error ?? new Error("read failed"));
  1618→    reader.readAsDataURL(file);
  1619→  });
  1620→}
